% Custom Heading Sizes (matching Word template)
% ==========================================

\newcommand{\sectionbreak}{\clearpage}

% Level 1 Heading (16pt) - e.g., \section
% The {16}{19} means 16pt font size with 19pt line spacing.
\titleformat{\section}
  {\fontsize{16}{19}\bfseries\sffamily}{\thesection}{1em}{}
\titlespacing*{\section}{0pt}{0pt}{0pt}

% Unnumbered Level 1 Heading (16pt) - Centered for Declaration, TOC, etc.
\titleformat{name=\section,numberless}
  {\fontsize{16}{19}\bfseries\sffamily\centering}{}{0em}{}
\titlespacing*{name=\section,numberless}{0pt}{0pt}{0pt}

% Level 2 Heading (14pt) - e.g., \subsection
\titleformat{\subsection}
  {\fontsize{14}{17}\bfseries\sffamily}{\thesubsection}{1em}{}
\titlespacing*{\subsection}{0pt}{0pt}{0pt}

% Level 3 Heading (12pt) - e.g., \subsubsection
\titleformat{\subsubsection}
  {\fontsize{12}{14}\bfseries\sffamily}{\thesubsubsection}{1em}{}
\titlespacing*{\subsubsection}{0pt}{0pt}{0pt}

% Level 4 Heading (11pt) - e.g., \paragraph
\titleformat{\paragraph}
  {\fontsize{11}{13}\bfseries\sffamily}{\theparagraph}{1em}{}
\titlespacing*{\paragraph}{0pt}{0pt}{0pt}

% --- Custom HEG Title Page ---
\newcommand{\makeHEGtitle}[5]{
% #1: Author Name
% #2: Supervisor Name(s)
% #3: Date/Place
% #4: HEG Logo Path
% #5: HES-SO Logo Path

\begin{titlepage}
    \begin{tikzpicture}[remember picture, overlay]
        % 1. Top Left: HEG Logo
        \node[anchor=north west, xshift=0.4cm, yshift=-0.9cm] at (current page.north west) {
            \includegraphics[width=0.3\textwidth]{#4}
        };
        % 2. Bottom Right: HES-SO Logo
        \node[anchor=south east, xshift=-1.7cm, yshift=0.8cm] at (current page.south east) {
            \includegraphics[width=0.25\textwidth]{#5}
        };

        % 3. Main Title — bottom of text at 5.8cm from top
        \node[anchor=south, yshift=-9.0cm] at (current page.north) {
            \parbox{0.8\textwidth}{\centering\linespread{1.5}\selectfont\fontsize{18}{22}\selectfont\bfseries \fulltitle}
        };

        % 4. Author — bottom of text at 14.1cm from top
        \node[anchor=south, yshift=-15.6cm] at (current page.north) {
            \parbox{0.8\textwidth}{\centering\linespread{1.2}\selectfont
                {\bfseries Master's thesis by:}\\[0.2cm]
                {\bfseries #1}}
        };

        % 5. Supervisor — bottom of text at 19.6cm from top
        \node[anchor=south, yshift=-19.9cm] at (current page.north) {
            \parbox{0.8\textwidth}{\centering\linespread{1.2}\selectfont
                {Under the direction of:}\\[0.2cm]
                {\bfseries #2}}
        };

        % 6. Place/Date — centered between Supervisor and School
        \node[anchor=center, yshift=-22.5cm] at (current page.north) {
            \parbox{0.8\textwidth}{\centering\linespread{1.2}\selectfont
                {\bfseries #3}}
        };

        % 7. School — bottom of text at 25.7cm from top
        \node[anchor=south, yshift=-26cm] at (current page.north) {
            \parbox{0.8\textwidth}{\centering\linespread{1.2}\selectfont
                {\bfseries Information Sciences}\\
                {\bfseries Haute École de Gestion de Genève (HEG-GE)}}
        };
    \end{tikzpicture}
\end{titlepage}
}
